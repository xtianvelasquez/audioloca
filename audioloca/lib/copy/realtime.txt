import 'dart:typed_data';
import 'dart:math' as math;
import 'package:logger/logger.dart';
import 'package:flutter/material.dart';
import 'package:camera/camera.dart';
import 'package:tflite_flutter/tflite_flutter.dart';
import 'package:image/image.dart' as img;
import 'package:google_mlkit_face_detection/google_mlkit_face_detection.dart';

final log = Logger();

class Tab3 extends StatefulWidget {
  const Tab3({super.key});
  @override
  State<Tab3> createState() => Tab3State();
}

class Tab3State extends State<Tab3> {
  CameraController? _cameraController;
  Interpreter? _interpreter;
  FaceDetector? _faceDetector;

  String _predictedEmotion = "Initializing...";
  double _confidence = 0.0;
  String _debugInfo = "";

  bool _isProcessing = false;
  bool _isInitialized = false;

  @override
  void initState() {
    super.initState();
    _initializeAll();
  }

  Future<void> _initializeAll() async {
    try {
      await _loadModel();
      await _initCamera();
      _initFaceDetector();
      _isInitialized = true;

      if (mounted) {
        setState(() {
          _predictedEmotion = "Ready - show your face";
        });
      }
    } catch (e, stackTrace) {
      log.e("Initialization error: $e $stackTrace");
      if (mounted) {
        setState(() {
          _predictedEmotion = "Error: $e";
        });
      }
    }
  }

  Future<void> _initCamera() async {
    try {
      final cameras = await availableCameras();

      // Find front camera
      CameraDescription? frontCamera;
      for (var camera in cameras) {
        if (camera.lensDirection == CameraLensDirection.front) {
          frontCamera = camera;
          break;
        }
      }

      if (frontCamera == null && cameras.isNotEmpty) {
        frontCamera = cameras.first;
      } else if (frontCamera == null) {
        throw Exception("No cameras available.");
      }

      _cameraController = CameraController(
        frontCamera,
        ResolutionPreset.low,
        enableAudio: false,
        imageFormatGroup: ImageFormatGroup.yuv420,
      );

      await _cameraController!.initialize();

      // Start frame processing with throttling
      _cameraController!.startImageStream((CameraImage image) {
        if (_isProcessing || !_isInitialized || _interpreter == null) return;
        _isProcessing = true;
        Future.delayed(Duration(milliseconds: 500), () {
          _processFrame(image).whenComplete(() => _isProcessing = false);
        });
      });

      if (mounted) setState(() {});
    } catch (e, stackTrace) {
      log.e("Camera error: $e");
      if (mounted) {
        setState(() {
          _predictedEmotion = "Camera error: $e $stackTrace";
        });
      }
    }
  }

  Future<void> _loadModel() async {
    try {
      final options = InterpreterOptions()
        ..threads = 2
        ..useNnApiForAndroid = true;

      _interpreter = await Interpreter.fromAsset(
        'assets/mobilenetv2_rafdb_finetuned.tflite',
        options: options,
      );

      log.i("Realtime model loaded successfully!");
    } catch (e, stackTrace) {
      log.e("Realtime model loading error: $e $stackTrace");
      rethrow;
    }
  }

  void _initFaceDetector() {
    _faceDetector = FaceDetector(
      options: FaceDetectorOptions(
        performanceMode: FaceDetectorMode.fast,
        enableContours: false,
        enableLandmarks: false,
        enableClassification: false,
        minFaceSize: 0.15,
      ),
    );
  }

  Future<void> _processFrame(CameraImage image) async {
    try {
      final InputImage inputImage = _buildInputImageFromCameraImage(image);
      final List<Face> faces = await _faceDetector!.processImage(inputImage);

      if (faces.isEmpty) {
        if (mounted) {
          setState(() {
            _predictedEmotion = "No face detected.";
            _confidence = 0.0;
            _debugInfo = "No faces found.";
          });
        }
        return;
      }

      final Face face = faces.reduce(
        (a, b) =>
            (a.boundingBox.width * a.boundingBox.height) >
                (b.boundingBox.width * b.boundingBox.height)
            ? a
            : b,
      );

      final Rect box = face.boundingBox;

      // 2. Convert to RGB and crop face
      final img.Image rgbImage = _convertYUV420ToImageRGB(image);

      // Add padding and ensure within bounds (similar to working code)
      final padding = 20.0;
      final double x = (box.left - padding).clamp(0, image.width - 1);
      final double y = (box.top - padding).clamp(0, image.height - 1);
      final double width = (box.width + 2 * padding).clamp(1, image.width - x);
      final double height = (box.height + 2 * padding).clamp(
        1,
        image.height - y,
      );

      final img.Image faceImage = img.copyCrop(
        rgbImage,
        x: x.toInt(),
        y: y.toInt(),
        width: width.toInt(),
        height: height.toInt(),
      );

      // 3. Preprocess EXACTLY like working code
      final List<List<List<List<double>>>> input = _preprocessImageWorking(
        faceImage,
      );

      // 4. Run inference
      final outputShape = _interpreter!.getOutputTensor(0).shape;
      final output = List.filled(
        outputShape.reduce((a, b) => a * b),
        0.0,
      ).reshape(outputShape);

      _interpreter!.run(input, output);

      // Convert output to list of doubles
      final List<double> predictions = List<double>.from(output[0]);

      // 5. Find top emotion
      int maxIdx = 0;
      double maxVal = predictions[0];

      for (int i = 1; i < predictions.length; i++) {
        if (predictions[i] > maxVal) {
          maxVal = predictions[i];
          maxIdx = i;
        }
      }

      // Apply softmax for probabilities
      double sumExp = 0.0;
      for (double val in predictions) {
        sumExp += math.exp(val);
      }
      final double confidence = sumExp > 0
          ? math.exp(predictions[maxIdx]) / sumExp
          : 0.0;

      log.i(
        "Prediction: ${_mapEmotion(maxIdx)} ($maxIdx), confidence=$confidence",
      );

      // 6. Update UI
      if (mounted) {
        setState(() {
          _predictedEmotion = _mapEmotion(maxIdx);
          _confidence = confidence;
          _debugInfo =
              "Face: ${faces.length} (${box.width.toInt()}x${box.height.toInt()})";
        });
      }
    } catch (e, stackTrace) {
      log.e("Frame processing error: $e $stackTrace");
      if (mounted) {
        setState(() {
          _debugInfo = "Error: ${e.toString()}";
        });
      }
    }
  }

  /// Preprocess image EXACTLY like the working code
  List<List<List<List<double>>>> _preprocessImageWorking(img.Image image) {
    // Resize to 224x224
    final img.Image resized = img.copyResize(image, width: 224, height: 224);

    // Create input tensor exactly like working code: [1, 224, 224, 3]
    final input = List.generate(
      1, // batch size
      (_) => List.generate(
        224, // height
        (y) => List.generate(224, (x) {
          final pixel = resized.getPixel(x, y);

          // EXACTLY like working code: normalize to [-1, 1]
          final double r = (pixel.r / 127.5) - 1.0;
          final double g = (pixel.g / 127.5) - 1.0;
          final double b = (pixel.b / 127.5) - 1.0;

          return [r, g, b];
        }),
      ),
    );

    return input;
  }

  InputImage _buildInputImageFromCameraImage(CameraImage image) {
    final InputImageRotation rotation = _rotationIntToImageRotation(
      _cameraController!.description.sensorOrientation,
    );

    final InputImageMetadata metadata = InputImageMetadata(
      size: Size(image.width.toDouble(), image.height.toDouble()),
      rotation: rotation,
      format: InputImageFormat.nv21,
      bytesPerRow: image.planes[0].bytesPerRow,
    );

    return InputImage.fromBytes(
      bytes: _convertYUV420ToNV21(image),
      metadata: metadata,
    );
  }

  Uint8List _convertYUV420ToNV21(CameraImage image) {
    final int width = image.width;
    final int height = image.height;
    final int ySize = width * height;

    final Uint8List nv21 = Uint8List(ySize + (ySize ~/ 2));
    final planeY = image.planes[0];
    final planeU = image.planes[1];
    final planeV = image.planes[2];

    // Copy Y plane
    nv21.setRange(0, ySize, planeY.bytes);

    // Interleave V and U planes
    int uvIndex = ySize;
    for (int row = 0; row < height ~/ 2; row++) {
      for (int col = 0; col < width ~/ 2; col++) {
        final int uIndex =
            (row * planeU.bytesPerRow) + (col * (planeU.bytesPerPixel ?? 1));
        final int vIndex =
            (row * planeV.bytesPerRow) + (col * (planeV.bytesPerPixel ?? 1));

        if (vIndex < planeV.bytes.length) {
          nv21[uvIndex++] = planeV.bytes[vIndex];
        }

        if (uIndex < planeU.bytes.length) {
          nv21[uvIndex++] = planeU.bytes[uIndex];
        }
      }
    }

    return nv21;
  }

  img.Image _convertYUV420ToImageRGB(CameraImage image) {
    final int width = image.width;
    final int height = image.height;

    final img.Image rgbImage = img.Image(width: width, height: height);

    final planeY = image.planes[0];
    final planeU = image.planes[1];
    final planeV = image.planes[2];

    final int yRowStride = planeY.bytesPerRow;
    final int uvRowStride = planeU.bytesPerRow;
    final int uvPixelStride = planeU.bytesPerPixel ?? 1;

    for (int y = 0; y < height; y++) {
      for (int x = 0; x < width; x++) {
        final int uvRow = y ~/ 2;
        final int uvCol = x ~/ 2;

        final int yIndex = y * yRowStride + x;
        final int uIndex = uvRow * uvRowStride + uvCol * uvPixelStride;
        final int vIndex = uvRow * planeV.bytesPerRow + uvCol * uvPixelStride;

        final int yVal = planeY.bytes[yIndex] & 0xFF;
        final int uVal = planeU.bytes[uIndex] & 0xFF;
        final int vVal = planeV.bytes[vIndex] & 0xFF;

        // Convert YUV to RGB with proper casting
        final int r = ((yVal + 1.402 * (vVal - 128)).clamp(0, 255)).toInt();
        final int g =
            ((yVal - 0.344136 * (uVal - 128) - 0.714136 * (vVal - 128)).clamp(
              0,
              255,
            )).toInt();
        final int b = ((yVal + 1.772 * (uVal - 128)).clamp(0, 255)).toInt();

        rgbImage.setPixelRgba(x, y, r, g, b, 255);
      }
    }

    return rgbImage;
  }

  InputImageRotation _rotationIntToImageRotation(int rotation) {
    switch (rotation) {
      case 0:
        return InputImageRotation.rotation0deg;
      case 90:
        return InputImageRotation.rotation90deg;
      case 180:
        return InputImageRotation.rotation180deg;
      case 270:
        return InputImageRotation.rotation270deg;
      default:
        return InputImageRotation.rotation0deg;
    }
  }

  String _mapEmotion(int idx) {
    const emotions = [
      "Angry", // 0
      "Disgust", // 1
      "Fear", // 2
      "Happy", // 3
      "Neutral", // 4
      "Sad", // 5
      "Surprise", // 6
    ];
    return idx >= 0 && idx < emotions.length ? emotions[idx] : "Unknown ($idx)";
  }

  @override
  void dispose() {
    _cameraController?.dispose();
    _interpreter?.close();
    _faceDetector?.close();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Real-time Emotion Detection')),
      body: _cameraController == null || !_cameraController!.value.isInitialized
          ? Center(
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  const CircularProgressIndicator(),
                  const SizedBox(height: 20),
                  Text(_predictedEmotion),
                  if (_debugInfo.isNotEmpty)
                    Padding(
                      padding: const EdgeInsets.only(top: 10),
                      child: Text(
                        _debugInfo,
                        style: const TextStyle(
                          fontSize: 12,
                          color: Colors.grey,
                        ),
                      ),
                    ),
                ],
              ),
            )
          : Stack(
              children: [
                CameraPreview(_cameraController!),
                Positioned(
                  bottom: 20,
                  left: 0,
                  right: 0,
                  child: Center(
                    child: Container(
                      padding: const EdgeInsets.all(16),
                      decoration: BoxDecoration(
                        color: Colors.black.withOpacity(0.7),
                        borderRadius: BorderRadius.circular(10),
                      ),
                      child: Column(
                        mainAxisSize: MainAxisSize.min,
                        children: [
                          Text(
                            _predictedEmotion,
                            style: const TextStyle(
                              color: Colors.white,
                              fontSize: 24,
                              fontWeight: FontWeight.bold,
                            ),
                          ),
                          const SizedBox(height: 4),
                          Text(
                            '${(_confidence * 100).toStringAsFixed(1)}%',
                            style: TextStyle(
                              color: Colors.white.withOpacity(0.8),
                              fontSize: 16,
                            ),
                          ),
                          if (_debugInfo.isNotEmpty) ...[
                            const SizedBox(height: 8),
                            Text(
                              _debugInfo,
                              style: TextStyle(
                                color: Colors.white.withOpacity(0.6),
                                fontSize: 12,
                              ),
                            ),
                          ],
                        ],
                      ),
                    ),
                  ),
                ),
              ],
            ),
    );
  }
}
